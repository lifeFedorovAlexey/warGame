// Offline authoring: only emits coordinates; never modifies terrain or m01.
import fs from 'node:fs';
import assert from 'node:assert/strict';
import {makeMission} from './m02-fixture.mjs';
import {installM02Config} from '../campaign/mission02-config.js';
const m=makeMission(),field=m.field;
installM02Config(m.config);
m.state.bounds=[0,0,112,127];m.rebuildNavigation();
const start=field.gridToWorld(m.config.startGrid),layout={version:1,bounds:m.state.bounds,bases:[],entrance:[]};
const sites=[
 ['east_factory',[86.5,93],['scav-factory','bunker'],[83.5,92.5]],
 ['hill_factory',[26.2,109.6],['scav-factory','bunker','bunker'],[28.5,108.5]],
 ['sensor_outpost',[101.5,110.3],['scav-sensor','bunker'],[100.5,106.5]],
 ['energy_hub',[79.3,118],['scav-factory','generator','bunker','bunker'],[79.5,113.5]],
];
function free(p,kind='bukhanka'){const pose={...p,kind,heading:0,owner:'scavs'};return m.vehicleMotionClear(pose,pose,pose);}
function pick(center,kind,index){
 const spec=m.config.buildings[kind],hw=spec.footprintMetres[0]*m.config.worldScale/2,hd=spec.footprintMetres[1]*m.config.worldScale/2;
 const candidates=[];
 for(let dy=-320;dy<=320;dy+=16)for(let dx=-320;dx<=320;dx+=16)candidates.push({x:center.x+dx,y:center.y+dy,score:dx*dx+dy*dy});
 candidates.sort((a,b)=>a.score-b.score||a.x-b.x||a.y-b.y);
 for(const p of candidates){
  if(m.state.buildings.some(b=>b.hp>0&&Math.abs(b.x-p.x)<b.halfWidth+hw+28&&Math.abs(b.y-p.y)<b.halfDepth+hd+28))continue;
  let ok=true;
  for(let y=-hd-10;y<=hd+10;y+=8)for(let x=-hw-10;x<=hw+10;x+=8){const g=field.gradientAt(p.x+x,p.y+y);if(Math.hypot(g.x,g.y)>.12||m.nav.roads.coverage(p.x+x,p.y+y)>.05)ok=false;}
  if(!ok)continue;
  const exit={x:p.x,y:p.y-45},pickup={x:p.x+hw+24,y:p.y};
  if(!free(exit)||!free(pickup)||!m.nav.findPath(start,pickup)?.length)continue;
  const b=m.addStructure(kind,'scavs',p,{id:'m02.'+index});m.rebuildNavigation();
  if(!free(exit)||!free(pickup)||!m.nav.findPath(start,pickup)?.length){m.state.buildings.pop();m.rebuildNavigation();continue;}
  return {id:b.id,kind,grid:field.worldToGrid(p.x,p.y),pickupGrid:field.worldToGrid(pickup.x,pickup.y)};
 }
 throw new Error('No placement '+index);
}
for(const [id,grid,kinds,gate] of sites){
 const center=field.gridToWorld(grid),base={id,grid,gateGrid:gate,structures:[]};
 for(const [i,kind] of kinds.entries())base.structures.push(pick(center,kind,`${id}.${kind}${i}`));
 const factory=base.structures.find(b=>b.kind==='scav-factory');
 const anchor=field.gridToWorld(factory?.grid??grid);
 const slots=[];
 for(let dy=-210;dy<=-60;dy+=32)for(let dx=-120;dx<=120;dx+=32){const p={x:anchor.x+dx,y:anchor.y+dy};if(free(p)&&m.nav.findPath(field.gridToWorld(gate),p)?.length)slots.push(field.worldToGrid(p.x,p.y));}
 assert(slots.length>=8,id+' assembly');base.rallyGrids=slots.slice(0,12);
 base.guardGrids=slots.slice(-5);
 layout.bases.push(base);
}
// The two guns flank the existing southern highway, with room to stage north.
for(const [i,grid] of [[36,85],[40,85]].entries())layout.entrance.push(pick(field.gridToWorld(grid),'bunker','entrance.bunker'+i));
layout.entranceGrid=[38,84];
layout.observer={grid:[81,106],exitGrid:[92,87]};
const desired=field.gridToWorld(layout.observer.exitGrid);
const exit=field.roadNetwork.paths.find(p=>p.to==='east-middle'&&p.from==='north-east').points.slice().sort((a,b)=>Math.hypot(a.x-desired.x,a.y-desired.y)-Math.hypot(b.x-desired.x,b.y-desired.y)).find(p=>free(p)&&m.nav.findPath(start,p)?.length);
assert(exit);layout.observer.exitGrid=field.worldToGrid(exit.x,exit.y);
for(const grid of [layout.observer.grid,layout.observer.exitGrid]){const p=field.gridToWorld(grid);assert(free(p)&&m.nav.findPath(start,p)?.length,'observer '+grid);}
fs.mkdirSync('assets/campaign/alpha/mission02',{recursive:true});
fs.writeFileSync('assets/campaign/alpha/mission02/layout.js','// Authored against the existing Alpha terrain; generated by tools/author-m02-layout.mjs.\nexport default '+JSON.stringify(layout,null,2)+';\n');
